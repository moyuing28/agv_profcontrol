// 隔离库存适配器：执行真实 CargoResources.cs / Models.cs，不加载宿主或驱动车辆。
global using System;
global using System.Linq;
global using System.Collections.Generic;

namespace ProfControl
{
    interface IStation { string Mark { get; } bool IsEmpty(); }
    public interface IAgv { }
    sealed class TestAgv : IAgv { }
    sealed class TestStation : IStation
    {
        public string Mark { get; set; }
        public bool IsEmpty() => !API.Cargos.Any(c => ReferenceEquals(c.Owner, this));
    }
    sealed class CargoInfo { public string Name = "cargo"; }
    sealed class ICargo
    {
        public CargoInfo Info = new CargoInfo();
        public string Tag;
        public int Quanlity = 1;
        public object Owner;
    }
    static class API
    {
        public static List<ICargo> Cargos = new List<ICargo>();
        public static object GetOwner(ICargo cargo) => cargo.Owner;
    }
    public partial class SampleProject
    {
        readonly object cargoBatchSync = new object();
        readonly List<CargoMoveBatch> cargoMoveBatches = new List<CargoMoveBatch>();
        readonly Dictionary<string, string> stationCargoAreas = new Dictionary<string, string>();
        sealed class CargoPocketConfig
        {
            public string Key;
            public string Error = null;
            public List<IStation> Slots;
        }
        readonly Dictionary<string, CargoPocketConfig> pockets = new Dictionary<string, CargoPocketConfig>();
        CargoPocketConfig GetCargoPocket(string key) => key != null && pockets.TryGetValue(key, out var p) ? p : null;
        string GetStationPocketKey(string mark) => pockets.Values.FirstOrDefault(p => p.Slots.Any(s => s.Mark == mark))?.Key;
        bool IsExpectedCargoAtStation(string mark, string name, string tag)
        {
            var cargos = API.Cargos.Where(c => c.Owner is IStation s && s.Mark == mark).ToList();
            return cargos.Count == 1 && cargos[0].Info.Name == name && cargos[0].Tag == tag && cargos[0].Quanlity == 1;
        }
        bool TryGetStationCargo(string mark, out ICargo cargo, out string error)
        {
            var matches = API.Cargos.Where(c => c.Owner is IStation s && s.Mark == mark).ToList();
            cargo = matches.FirstOrDefault();
            var candidate = cargo;
            error = matches.Count == 1 && cargo.Quanlity == 1 && !string.IsNullOrWhiteSpace(cargo.Tag) &&
                API.Cargos.Count(c => c.Info.Name == candidate.Info.Name && c.Tag == candidate.Tag) == 1 ? null : "invalid cargo";
            return error == null;
        }
        static SampleProject Fixture()
        {
            API.Cargos.Clear();
            var p = new SampleProject();
            foreach (var key in new[] { "A", "B", "C", "D" })
            {
                var slots = Enumerable.Range(0, 3).Select(i => (IStation)new TestStation { Mark = key + i }).ToList();
                p.pockets.Add(key, new CargoPocketConfig { Key = key, Slots = slots });
                foreach (var s in slots) p.stationCargoAreas.Add(s.Mark, "area");
            }
            p.Put("a0", "A0"); p.Put("a1", "A1"); p.Put("b0", "B0"); p.Put("c0", "C0");
            return p;
        }
        IStation Station(string mark) => pockets.Values.SelectMany(p => p.Slots).Single(s => s.Mark == mark);
        void Put(string tag, string mark) => API.Cargos.Add(new ICargo { Tag = tag, Owner = Station(mark) });
        static ICargo Cargo(string tag) => API.Cargos.Single(c => c.Tag == tag);
        static TaskInfo Move(string tag = "a1", string from = "A1", string to = "B1") =>
            new TaskInfo { CargoName = "cargo", CargoTag = tag, FromMark = from, ToMark = to };
        static int assertions;
        static void Check(bool ok, string name)
        {
            if (!ok) throw new Exception("FAIL: " + name);
            assertions++;
        }
        static bool Can(SampleProject p, string tag, string target) =>
            p.CanPointCargoMove("cargo", tag, target, null, out _, out _);
        public static void Main()
        {
            var p = Fixture();
            Check(p.TryGetPointPocketState("D", out var empty, out _) && empty.OutermostMark == null && empty.NextDropMark == "D0", "empty lane starts at deepest");
            Check(p.TryGetPointPocketState("A", out var a, out _) && a.OutermostMark == "A1" && a.NextDropMark == "A2", "live outermost and next drop");
            Check(Can(p, "a1", "B1"), "legal cross lane move");
            Check(!Can(p, "a0", "B1"), "blocked source");
            Check(!Can(p, "a1", "B2"), "cannot skip empty depth");
            Check(!Can(p, "a1", "B0"), "occupied destination");
            Check(!Can(p, "a1", "A1"), "already there requires finalize");
            Check(!Can(p, "a1", "A2") && !Can(p, "a1", "A0"), "same lane checks hypothetical pickup");
            p.Put("d2", "D2");
            Check(!p.TryGetPointPocketState("D", out _, out var hole) && hole.Contains("No-hole"), "invalid hole rejected");
            p = Fixture(); p.Put("a2", "A2");
            Check(p.TryGetPointPocketState("A", out var full, out _) && full.NextDropMark == null, "full lane");
            p = Fixture(); p.Put("extra", "A1");
            Check(!p.TryGetPointPocketState("A", out _, out _), "multiple cargo at one slot");
            p = Fixture(); Cargo("a1").Owner = p.Station("D0");
            Check(p.TryGetPointCargoSource("cargo", "a1", out var live, out _) && live.Mark == "D0", "query relocated live source");
            Check(!p.TryReservePointMove(Move(), out _) && p.pointMoveReservations.Count == 0, "stale source fails atomically");
            p = Fixture(); Cargo("a1").Owner = new TestAgv();
            Check(!p.TryGetPointCargoSource("cargo", "a1", out _, out _), "carried cargo is not a pickup source");
            p = Fixture(); p.Put("a1", "D0");
            Check(!p.TryGetPointCargoSource("cargo", "a1", out _, out _), "duplicate identity rejected");
            p = Fixture(); p.cargoMoveBatches.Add(new CargoMoveBatch { MoveType = CargoMoveType.PointToPoint,
                Items = new List<CargoMoveItem> { new CargoMoveItem { CargoName = "cargo", CargoTag = "a1", Status = CargoTaskStatus.Completed } } });
            Check(!Can(p, "a1", "B1"), "completed customer cargo fixed");
            p = Fixture(); p.stationCargoAreas.Remove("B2");
            Check(!Can(p, "a1", "B1"), "invalid access slot configuration");

            p = Fixture(); var first = Move();
            Check(p.TryReservePointMove(first, out _) && p.TryReservePointMove(first, out _) && p.pointMoveReservations.Count == 1, "idempotent acquisition");
            Check(!p.TryReservePointMove(Move("c0", "C0", "B1"), out _) && p.pointMoveReservations.Count == 1, "same destination conflict without partial reservation");
            Check(Can(p, "c0", "D0"), "failed reservation does not hold its source");
            Check(!p.TryReservePointMove(Move("c0", "C0", "A2"), out _), "source outer access reservation blocks other drop");
            var impostor = Move(); impostor.TaskId = first.TaskId;
            Check(!p.ReleasePointSourceReservation(impostor) && !p.ReleasePointMoveReservation(impostor), "same id cannot steal another owner resources");
            Check(!p.TryReservePointMove(impostor, out _), "same id cannot replace owner");
            Check(!p.CanDropReservedPointMove(first, out _), "cannot drop before pickup");
            Check(p.ReleasePointSourceReservation(first), "release source reservation");
            Check(!p.CanDropReservedPointMove(first, out _), "drop checks actual carried cargo");
            first.WorkAgv = new TestAgv(); Cargo("a1").Owner = first.WorkAgv;
            Check(p.CanDropReservedPointMove(first, out _), "valid loaded drop after pickup");
            Check(!p.TryReservePointMove(Move("c0", "C0", "B1"), out _), "target retained after source release");
            Check(p.TryReservePointMove(Move("c0", "C0", "A1"), out _), "source access becomes available after pickup");
            p.Put("b1", "B1");
            Check(!p.CanDropReservedPointMove(first, out _), "drop rechecks live target inventory");
            API.Cargos.Remove(Cargo("b1"));
            first.ToMark = "D0";
            Check(!p.CanDropReservedPointMove(first, out _), "mutated action rejected");
            first.ToMark = "B1";
            Check(p.ReleasePointMoveReservation(first) && !p.ReleasePointMoveReservation(first), "owner release and repeat release safe");
            Check(p.TryReservePointMove(Move("b0", "B0", "D0"), out _), "target access available after final release");
            p = Fixture();
            var contenders = new[] { Move(), Move("c0", "C0", "B1") };
            var accepted = new bool[2];
            System.Threading.Tasks.Parallel.For(0, 2, i => accepted[i] = p.TryReservePointMove(contenders[i], out _));
            Check(accepted.Count(x => x) == 1 && p.pointMoveReservations.Count == 1, "concurrent claims have one winner");
            Console.WriteLine("PASS: " + assertions + " assertions (isolated inventory adapter; no host execution)");
        }
    }
}
